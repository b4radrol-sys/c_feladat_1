void elmozdulas(string irany)
{
    int x = 0;
    int y = 0;


    foreach (char c in irany)
    {
        if (c == 'J')
        {
            x++;
        }
        else if (c == 'B')
        {
            x--;
        }
        else if (c == 'F')
        {
            y++;
        }
        else if (c == 'L')
        {
            y--;
        }
    }

    string xirany = "";
    string yirany = "";

    if (x > 0)
    {
        xirany = "jobbra";
    }
    else
    {
        xirany = "balra";
    }

    if (y > 0)
    {
        yirany = "fel";
    }
    else
    {
        yirany = "le";
    }

    if (x != 0)
    {
        if (y != 0)
        {
            Console.WriteLine($"{Math.Abs(x)} lepes {xirany}, {Math.Abs(y)} lepes {yirany}");
        }
        else
        {
            Console.WriteLine($"{Math.Abs(x)} lepes {xirany}");
        }

    }
    else
    {
        if (y != 0)
        {
            Console.WriteLine($"{Math.Abs(y)} lepes {yirany}");
        }
        else
        {
            Console.WriteLine("Nem mentunk sehova");
        }
    }
    

}

elmozdulas(Console.ReadLine());
