int fibo(int n)
{
    int el = 0;
    int mo = 1;
    int ossz = 0;

    for (int i = 1; i < n; i++)
    {
        ossz = el + mo;
        el = mo;
        mo = ossz;
    }

    return mo;
}

Console.WriteLine(fibo(42));
